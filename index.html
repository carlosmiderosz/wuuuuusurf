<script>

const price = 10;

let cart = [];


const videos = [

  {id:"#001", url:"https://www.youtube.com/embed/VIDEO1"},

  {id:"#002", url:"https://www.youtube.com/embed/VIDEO2"},

  {id:"#003", url:"https://www.youtube.com/embed/VIDEO3"}

];


const grid = document.getElementById("grid");


videos.forEach(v => {

  let div = document.createElement("div");

  div.className = "card";

  div.innerHTML = `

    <div class="id">${v.id}</div>

    <div class="check"></div>

    <iframe src="${v.url}" frameborder="0"></iframe>

  `;


  div.onclick = () => toggle(v.id, div);

  grid.appendChild(div);

});


function toggle(id, el){

  if(cart.includes(id)){

    cart = cart.filter(x => x !== id);

    el.classList.remove("selected");

  } else {

    cart.push(id);

    el.classList.add("selected");

  }

  update();

}


function update(){

  document.getElementById("count").innerText = cart.length;

  document.getElementById("total").innerText = cart.length * price;

}


function checkout(){

  if(cart.length === 0){

    alert("Selecciona al menos 1 clip");

    return;

  }


  let mensaje = `Hola, quiero estos clips:\n${cart.join(", ")}\nTotal: S/${cart.length * price}`;

  

  window.open(`https://wa.me/51999999999?text=${encodeURIComponent(mensaje)}`);

}
